// Java Code: Two Named Connection Pools


import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;

import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.ResultSet;
import java.sql.Statement;

public class MultiPoolExample {

    public static void main(String[] args) {
        // Pool 1: testdb
        HikariConfig config1 = new HikariConfig();
        config1.setPoolName("TestDBPool");
        config1.setJdbcUrl("jdbc:mysql://localhost:3306/mydb");
        config1.setUsername("root");
        config1.setPassword("root");

        HikariDataSource testDbPool = new HikariDataSource(config1);

        // Pool 2: inventorydb
        HikariConfig config2 = new HikariConfig();
        config2.setPoolName("mypool1");
        config2.setJdbcUrl("jdbc:mysql://localhost:3306/mydb");
        config2.setUsername("root");
        config2.setPassword("root");

        HikariDataSource inventoryDbPool = new HikariDataSource(config2);

        // Use first pool
        runQuery(testDbPool, "SELECT * FROM users");

        // Use second pool
        runQuery(inventoryDbPool, "SELECT * FROM products");

        // Close pools
        testDbPool.close();
        inventoryDbPool.close();
    }

    private static void runQuery(DataSource dataSource, String query) {
        try (Connection conn = dataSource.getConnection();
             Statement stmt = conn.createStatement();
             ResultSet rs = stmt.executeQuery(query)) {

            System.out.println("Results from pool: " + ((HikariDataSource) dataSource).getPoolName());
            int colCount = rs.getMetaData().getColumnCount();

            while (rs.next()) {
                for (int i = 1; i <= colCount; i++) {
                    System.out.print(rs.getString(i) + "\t");
                }
                System.out.println();
            }

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
