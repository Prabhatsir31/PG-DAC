package mypack;

import org.hibernate.Session;
import org.hibernate.SessionFactory;
import org.hibernate.Transaction;


public class Main 
{
	public static void main(String[] args) 
	{
		SessionFactory factory=MyConfig.getSessionFactory();
		Session session=factory.openSession();
		Transaction tr=session.beginTransaction();
        Employee e1=new Employee();
        e1.setName("abc");
        e1.setDesig("officer");
        e1.setSalary(30000);
            
        Employee e2=new Employee();       
        e2.setName("xyz");
        e2.setDesig("manager");
        e2.setSalary(40000);
        session.persist(e1);
        session.persist(e2);
        tr.commit(); 
        System.out.println("Let's close the session");
        session.close();
        session=factory.openSession();
        System.out.println("Let's retrieve both the records after using new session");
        Employee ref1=(Employee)session.get(Employee.class,e1.getEmpid());
        System.out.println(ref1);
        if(e1==ref1)
        {
        	System.out.println("e1 and ref1 are refering to same object");
        }
        else
        {
           	System.out.println("e1 and ref1 are not refering to same object");
        }
        Employee ref2=(Employee)session.get(Employee.class,e2.getEmpid());
        System.out.println(ref2);
        if(e2==ref2)
        {
        	System.out.println("e2 and ref2 are refering to same object");
        }
        else
        {
           	System.out.println("e2 and ref2 are not refering to same object");
        }
        factory.close();
        System.out.println("done with employees");  
	}
}
